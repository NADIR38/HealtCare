$baseUrl = "https://localhost:7227" # Update with your port
$outputFile = "api-test-results-$(Get-Date -Format 'yyyy-MM-dd-HHmmss').txt"

# Color output functions - Using ASCII characters only
function Write-TestSuccess { 
    param($message) 
    Write-Host "[PASS] $message" -ForegroundColor Green 
}

function Write-TestFailure { 
    param($message) 
    Write-Host "[FAIL] $message" -ForegroundColor Red 
}

function Write-TestInfo { 
    param($message) 
    Write-Host "[INFO] $message" -ForegroundColor Cyan 
}

function Write-TestWarning { 
    param($message) 
    Write-Host "[WARN] $message" -ForegroundColor Yellow 
}

# Initialize results
$results = @{
    Total = 0
    Passed = 0
    Failed = 0
    Skipped = 0
    Tests = @()
}

    Doctor = @{ Email = "doctor@healthcare.com"; Password = "Doctor@123"; Token = $null }
    Patient = @{ Email = "patient@healthcare.com"; Password = "Patient@123"; Token = $null }
}

function Test-Endpoint {
    param(
        [string]$Name,
        [string]$Method,
        [string]$Endpoint,
        [string]$Token = $null,
        [object]$Body = $null,
        [int]$ExpectedStatus = 200
    )
    
    $results.Total++
    
    try {
        $headers = @{ "Content-Type" = "application/json" }
        if ($Token) {
            $headers["Authorization"] = "Bearer $Token"
        }
        
        $params = @{
            Uri = "$baseUrl$Endpoint"
            Method = $Method
            Headers = $headers
            TimeoutSec = 30
        }
        
        if ($Body) {
            $params["Body"] = ($Body | ConvertTo-Json -Depth 10)
        }
        
        # Use try-catch to capture status code for both success and error responses
        try {
            $response = Invoke-RestMethod @params
            $statusCode = 200 # Default success code
            
            # Try to get actual status code if available
            if ($response.PSObject.Properties['StatusCode']) {
                $statusCode = $response.StatusCode
            }
        }
        catch {
            # Extract status code from error
            if ($_.Exception.Response) {
                $statusCode = [int]$_.Exception.Response.StatusCode
            }
            else {
                throw
            }
            $response = $null
        }
        
        if ($statusCode -eq $ExpectedStatus) {
            Write-TestSuccess "$Name - Status: $statusCode"
            $results.Passed++
            $results.Tests += @{
                Name = $Name
                Status = "PASSED"
                StatusCode = $statusCode
                Endpoint = "$Method $Endpoint"
            }
            return $response
        } else {
            Write-TestFailure "$Name - Expected: $ExpectedStatus, Got: $statusCode"
            $results.Failed++
            $results.Tests += @{
                Name = $Name
                Status = "FAILED"
                StatusCode = $statusCode
                Endpoint = "$Method $Endpoint"
                Error = "Status code mismatch"
            }
            return $null
        }
    }
    catch {
        Write-TestFailure "$Name - $($_.Exception.Message)"
        $results.Failed++
        $results.Tests += @{
            Name = $Name
            Status = "FAILED"
            Endpoint = "$Method $Endpoint"
            Error = $_.Exception.Message
        }
        return $null
    }
}
